CREATE DATABASE IF NOT EXISTS clinica;

USE clinica;

CREATE TABLE IF NOT EXISTS convenio (
    id_convenio INT NOT NULL AUTO_INCREMENT,
    nome VARCHAR(100) NOT NULL,
    telefone VARCHAR(20),
    percentual_desconto DECIMAL(5,2) DEFAULT 0.00,
    PRIMARY KEY (id_convenio)
);

CREATE TABLE IF NOT EXISTS paciente (
    id_paciente INT NOT NULL AUTO_INCREMENT,
    nome VARCHAR(150) NOT NULL,
    cpf VARCHAR(14) NOT NULL UNIQUE,
    data_nacimento DATE NOT NULL,
    telefone VARCHAR(20),
    Email VARCHAR(150),
    endereco VARCHAR(200),
    cidade VARCHAR(100),
    id_convenio INT,
    PRIMARY KEY (id_paciente),
    FOREIGN KEY (id_convenio)
        REFERENCES convenio(id_convenio)
);

CREATE TABLE IF NOT EXISTS medico (
    id_medico INT NOT NULL AUTO_INCREMENT,
    nome VARCHAR(150) NOT NULL,
    crm VARCHAR(20) NOT NULL UNIQUE,
    especialidade VARCHAR(100) NOT NULL,
    telefone VARCHAR(20),
    Email VARCHAR(150),
    valor_consulta DECIMAL(10,2),
    foto VARCHAR(255),
    PRIMARY KEY (id_medico)
);

CREATE TABLE IF NOT EXISTS consulta (
    id_consulta INT NOT NULL AUTO_INCREMENT,
    id_paciente INT NOT NULL,
    id_medico INT NOT NULL,
    data_consulta DATE NOT NULL,
    horario TIME NOT NULL,
    motivo VARCHAR(200),
    observacoes TEXT,
    status VARCHAR(30) NOT NULL DEFAULT 'Agendada',
    PRIMARY KEY (id_consulta),
    FOREIGN KEY (id_paciente)
        REFERENCES paciente(id_paciente),
    FOREIGN KEY (id_medico)
        REFERENCES medico(id_medico)
);

CREATE TABLE IF NOT EXISTS exame (
    id_exame INT NOT NULL AUTO_INCREMENT,
    id_paciente INT NOT NULL,
    id_consulta INT DEFAULT NULL,
    nome VARCHAR(100) NOT NULL,
    data_solicitacao DATE NOT NULL,
    data_realizacao DATE DEFAULT NULL,
    resultado VARCHAR(500),
    status_ VARCHAR(30) NOT NULL DEFAULT 'solicitado',
    valor DECIMAL(10,2),
    PRIMARY KEY (id_exame),
    FOREIGN KEY (id_paciente)
        REFERENCES paciente(id_paciente),
    FOREIGN KEY (id_consulta)
        REFERENCES consulta(id_consulta)
);

USE clinica;

SHOW TABLES;

SELECT * FROM paciente;

SELECT * FROM medico;

SELECT * FROM convenio;

SELECT
    c.id_consulta,
    p.nome AS paciente,
    m.nome AS medico,
    m.especialidade,
    c.data_consulta,
    c.horario,
    c.status
FROM consulta c
INNER JOIN paciente p
    ON c.id_paciente = p.id_paciente
INNER JOIN medico m
    ON c.id_medico = m.id_medico
ORDER BY c.data_consulta, c.horario;

SELECT
    p.nome AS paciente,
    p.telefone,
    p.cidade,
    cv.nome AS convenio,
    cv.percentual_desconto
FROM paciente p
LEFT JOIN convenio cv
    ON p.id_convenio = cv.id_convenio;

SELECT
    e.id_exame,
    e.nome AS exame,
    p.nome AS paciente,
    e.data_solicitacao,
    e.data_realizacao,
    e.status_,
    e.valor
FROM exame e
INNER JOIN paciente p
    ON e.id_paciente = p.id_paciente;

SELECT
    e.nome AS exame,
    p.nome AS paciente,
    c.data_consulta,
    c.horario,
    e.status_
FROM exame e
INNER JOIN paciente p
    ON e.id_paciente = p.id_paciente
LEFT JOIN consulta c
    ON e.id_consulta = c.id_consulta;

SELECT
    m.nome AS medico,
    COUNT(c.id_consulta) AS total_consultas
FROM medico m
LEFT JOIN consulta c
    ON m.id_medico = c.id_medico
GROUP BY m.id_medico, m.nome
ORDER BY total_consultas DESC;

SELECT *
FROM consulta
WHERE status = 'Agendada'
ORDER BY data_consulta, horario;
